-- Query 1: Row counts per Tier B table
select
  'accounts'         as tbl, count(*) from accounts         union all
  select 'ces_history',      count(*) from ces_history       union all
  select 'health_history',   count(*) from health_history    union all
  select 'activity_log',     count(*) from activity_log      union all
  select 'milestones',       count(*) from milestones        union all
  select 'stakeholders',     count(*) from stakeholders      union all
  select 'tasks',            count(*) from tasks             union all
  select 'surveys',          count(*) from surveys           union all
  select 'survey_responses', count(*) from survey_responses  union all
  select 'onboarding_plans', count(*) from onboarding_plans  union all
  select 'onboarding_tasks', count(*) from onboarding_tasks  union all
  select 'usage_history',    count(*) from usage_history     union all
  select 'churn_events',     count(*) from churn_events      union all
  select 'integrations',     count(*) from integrations      union all
  select 'automation_rules', count(*) from automation_rules  union all
  select 'automation_log',   count(*) from automation_log
order by tbl;

-- Query 2: Count of auth.users (how many users will get personal orgs)
select count(*) as total_users from auth.users;

-- Query 3: Verify actual automation table name
select tablename from pg_tables
where schemaname = 'public' and tablename like 'automat%';
